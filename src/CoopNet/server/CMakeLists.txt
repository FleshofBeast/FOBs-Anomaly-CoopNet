cmake_minimum_required(VERSION 3.21)
project(CoopNetServer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "The initial dedicated server uses Windows checkpoint and crypto APIs")
endif()
find_package(GameNetworkingSockets CONFIG REQUIRED)
add_executable(CoopNetServer ServerMain.cpp)
target_compile_definitions(CoopNetServer PRIVATE STEAMNETWORKINGSOCKETS_OPENSOURCE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(CoopNetServer PRIVATE GameNetworkingSockets::GameNetworkingSockets bcrypt)
if(MSVC)
  target_compile_options(CoopNetServer PRIVATE /W4 /WX /EHsc)
endif()
