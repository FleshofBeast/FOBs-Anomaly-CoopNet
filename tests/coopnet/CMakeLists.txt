cmake_minimum_required(VERSION 3.21)
project(CoopNetTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(GameNetworkingSockets CONFIG REQUIRED)
add_executable(GnsSmoke GnsSmoke.cpp)
target_link_libraries(GnsSmoke PRIVATE GameNetworkingSockets::GameNetworkingSockets)
target_compile_definitions(GnsSmoke PRIVATE STEAMNETWORKINGSOCKETS_OPENSOURCE)
if(MSVC)
  target_compile_options(GnsSmoke PRIVATE /W4 /WX /EHsc)
endif()
add_executable(DedicatedNetwork DedicatedNetwork.cpp)
target_link_libraries(DedicatedNetwork PRIVATE GameNetworkingSockets::GameNetworkingSockets)
target_compile_definitions(DedicatedNetwork PRIVATE STEAMNETWORKINGSOCKETS_OPENSOURCE)
if(MSVC)
  target_compile_options(DedicatedNetwork PRIVATE /W4 /WX /EHsc)
endif()
